// ====== KONFIGURASI ====== const SPREADSHEET_ID = '1LB_HDk-2ZJO1wGMZtghW-KawkGqmpFwSuG0A0y87v_A'; const SHEET_NAME = 'Sheet1'; const FOLDER_NAME = 'Foto Absen Online'; // ====== HANDLER POST ====== function doPost(e) { try { const data = JSON.parse(e.postData.contents); let fotoUrl = '-'; if (data.fotoBase64 && data.fotoBase64.indexOf('data:image') === 0) { const folder = getOrCreateFolder(FOLDER_NAME); const contentType = data.fotoBase64.split(',')[0].split(':')[1].split(';')[0]; const bytes = Utilities.base64Decode(data.fotoBase64.split(',')[1]); const blob = Utilities.newBlob(bytes, contentType, data.fotoName || ('foto_' + Date.now() + '.jpg')); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fotoUrl = file.getUrl(); } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME) || ss.getActiveSheet(); sheet.appendRow([ data.tanggal, data.nama, data.gudang, data.tipe, data.status, data.jamMasuk || '', data.jamPulang || '', data.lembur, fotoUrl ]); return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Absen tersimpan' })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ====== HANDLER GET ====== function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'API Absen aktif' })) .setMimeType(ContentService.MimeType.JSON); } // ====== HELPER ====== function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); return folders.hasNext() ? folders.next() : DriveApp.createFolder(name); }
